const firebaseConfig = {

  // projectId: "...",
  // storageBucket: "...",
  // messagingSenderId: "...",
  // appId: "..."
};

// ---- INIT ----
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);

// ---- AUTH PANEL WIRING (email/password) ----
const el = (s) => document.querySelector(s);
const authPanel   = el('#authPanel');
const openAuthBtn = el('#openAuthBtn');
const signOutBtn  = el('#signOutBtn');
const authStatus  = el('#authStatus');

const signInBtn      = el('#signInBtn');
const loginEmail     = el('#loginEmail');
const loginPassword  = el('#loginPassword');

const createBtn       = el('#createAccountBtn');
const createEmail     = el('#createEmail');
const createPassword  = el('#createPassword');
const createPassword2 = el('#createPassword2');

openAuthBtn?.addEventListener('click', () => {
  authPanel.style.display = (authPanel.style.display === 'none' || !authPanel.style.display) ? 'block' : 'none';
});
signOutBtn?.addEventListener('click', async () => { await signOut(auth); });

signInBtn?.addEventListener('click', async () => {
  try { await signInWithEmailAndPassword(auth, loginEmail.value.trim(), loginPassword.value); }
  catch (e) { alert(`Sign-in failed: ${e.message}`); }
});
createBtn?.addEventListener('click', async () => {
  try {
    if (createPassword.value !== createPassword2.value) throw new Error('Passwords do not match');
    await createUserWithEmailAndPassword(auth, createEmail.value.trim(), createPassword.value);
    authPanel.style.display = 'none';
  } catch (e) { alert(`Create failed: ${e.message}`); }
});

// Small helper for sub-tabs inside the auth card
document.querySelectorAll('.subtab-btn').forEach(b=>{
  b.addEventListener('click', ()=>{
    const id=b.dataset.subtab;
    document.querySelectorAll('.subtab-btn').forEach(x=>x.classList.remove('active'));
    document.querySelectorAll('.subtab').forEach(x=>x.classList.remove('active'));
    b.classList.add('active');
    document.getElementById(id)?.classList.add('active');
  });
});

// ---- REALTIME LISTENER FOR USER’S DOCUMENTS ----
let unsubRows = null;
function startRowsListener(uid) {
  if (unsubRows) { unsubRows(); unsubRows = null; }
  const q = query(
    collection(db, "farm_records"),
    where("uid", "==", uid),
    orderBy("date", "desc"),
    limit(1000) // you mentioned up to 1000 for testing
  );
  unsubRows = onSnapshot(q, (snap) => {
    const rows = snap.docs.map(d => {
      const data = d.data() || {};
      // normalize legacy: plot -> field
      if (data.plot && !data.field) data.field = data.plot;
      return { id: d.id, d: data };
    });
    document.dispatchEvent(new CustomEvent('cloud-rows', { detail: rows }));
  });
}

onAuthStateChanged(auth, (user) => {
  if (unsubRows) { unsubRows(); unsubRows = null; }
  if (user) {
    authStatus.textContent = `Signed in: ${user.email}`;
    signOutBtn.style.display = '';
    openAuthBtn.style.display = 'none';
    authPanel.style.display = 'none';
    startRowsListener(user.uid);
  } else {
    authStatus.textContent = 'Signed out';
    signOutBtn.style.display = 'none';
    openAuthBtn.style.display = '';
    document.dispatchEvent(new CustomEvent('cloud-rows', { detail: [] }));
  }
  document.dispatchEvent(new CustomEvent('auth-changed', { detail: user }));
});

// ---- WRITES (used by app.js) ----
export async function saveRecord(formValues) {
  const user = auth.currentUser;
  if (!user) throw new Error("Not signed in");
  const {
    id, date, field, category, material, quantity, materialRate,
    normalLab, specialLab, cost, notes
  } = formValues;

  const payload = {
    uid: user.uid,
    date, field, category,
    material: material || "",
    quantity: Number(quantity || 0),
    materialRate: materialRate != null && materialRate !== '' ? Number(materialRate) : null,
    normalLab: Number(normalLab || 0),
    specialLab: Number(specialLab || 0),
    cost: Number(cost || 0),
    notes: notes || "",
    updatedAt: serverTimestamp(),
  };

  if (id) {
    await updateDoc(doc(db, "farm_records", id), payload);
  } else {
    await addDoc(collection(db, "farm_records"), { ...payload, createdAt: serverTimestamp() });
  }
}

export async function deleteRecord(id) {
  const user = auth.currentUser;
  if (!user) throw new Error("Not signed in");
  await deleteDoc(doc(db, "farm_records", id));
}
